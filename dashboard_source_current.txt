   140	        async function fetchTelemetry() {
   141	            try {
   142	                const res = await fetch('/api/telemetry');
   143	                const data = await res.json();
   144	                
   145	                document.getElementById('btc-price').innerText = '$' + data.btc_price;
   146	                document.getElementById('total-equity').innerText = '$' + data.total_equity;
   147	                document.getElementById('reserve-vault').innerText = '$' + data.reserve_usdt;
   148	                document.getElementById('win-rate-stat').innerText = `${data.win_rate} (${data.total_profit_usdt})`;
   149	                
   150	                document.getElementById('rsi-val').innerText = data.rsi;
   151	                document.getElementById('trend-val').innerText = data.trend;
   152	
   153	                const pos = data.position;
   154	                if (pos.side !== 'FLAT') {
   155	                    document.getElementById('active-position').innerText = `${pos.side} (${pos.amount} BTC @ ${pos.entry_price})`;
   156	                    const pnlVal = parseFloat(pos.pnl);
   157	                    const pnlElem = document.getElementById('pos-pnl');
   158	                    pnlElem.innerText = `PnL: ${pnlVal >= 0 ? '+' : ''}${pnlVal.toFixed(2)} USDT [TP: ${pos.tp} | SL: ${pos.sl}]`;
   159	                    pnlElem.className = `text-xs font-bold mt-1 ${pnlVal >= 0 ? 'text-emerald-400' : 'text-rose-400'}`;
   160	                } else {
   161	                    document.getElementById('active-position').innerText = 'بدون معامله باز (در حال تحلیل)';
   162	                    document.getElementById('pos-pnl').innerText = 'PnL: $0.00';
   163	                    document.getElementById('pos-pnl').className = 'text-xs font-bold mt-1 text-gray-500';
   164	                }
   165	
   166	                // Update Chart
   167	                const currPrice = parseFloat(data.btc_price.replace(/,/g, ''));
   168	                priceChart.data.datasets[0].data.push(currPrice);
   169	                priceChart.data.datasets[0].data.shift();
   170	                priceChart.update('none');
   171	
   172	                // Update Trades
   173	                const tradesContainer = document.getElementById('trades-container');
   174	                if (data.recent_trades.length > 0) {
   175	                    tradesContainer.innerHTML = data.recent_trades.map(t => `
   176	                        <div class="flex justify-between border-b border-gray-800 pb-1">
   177	                            <span>[${t.time}] ${t.action}</span>
   178	                            <span class="${t.pnl.includes('+') ? 'text-emerald-400' : 'text-rose-400'} font-bold">${t.pnl}</span>
   179	                        </div>
   180	                    `).join('');
   181	                }
   182	
   183	                // Update Logs
   184	                const logsContainer = document.getElementById('logs-container');
   185	                logsContainer.innerHTML = data.logs.slice(-15).reverse().map(l => `<div>${l}</div>`).join('');
   186	
   187	            } catch (err) {
   188	                console.error("Telemetry fetch error:", err);
   189	            }
   190	        }

   192	        async function triggerAction(action) {
   193	            await fetch('/api/action', {
   194	                method: 'POST',
   195	                headers: { 'Content-Type': 'application/json' },
   196	                body: JSON.stringify({ action: action, amount: 0.02 })
   197	            });
   198	            fetchTelemetry();
   199	        }

   202	        fetchTelemetry();
