   256	            str(symbol).upper(): float(price)
   257	            for symbol, price in prices.items()
   258	        }

   260	        self.position_tracker.update_prices(normalized)

   262	    def get_equity(self):
   263	        equity = float(self.current_capital)
   264	        seen = set()
   265	
   266	        for position in self.position_tracker.positions.values():
   267	            if id(position) in seen:
   268	                continue
   269	            seen.add(id(position))
   270	
   271	            if not position.is_open:
   272	                continue
   273	
   274	            market_price = self.market_prices.get(
   275	                position.symbol, position.current_price
   276	            )
   277	
   278	            # در BUY هنگام ورود، cost از current_capital کسر شده؛
   279	            # برای Equity باید ارزش فعلی دارایی اضافه شود.
   280	            if position.side == "BUY":
   281	                equity += position.size * float(market_price)
   282	            else:
   283	                # مدل ساده short: collateral ورود + سود/زیان.
   284	                equity += (
   285	                    position.size * position.entry_price
   286	                    + position.size * (position.entry_price - float(market_price))
   287	                )
   288	
   289	        return equity

   317	            symbol=symbol,
   318	            side=side_text,
   319	            quantity=quantity,
   320	            current_price=price,
   321	            sl=sl if sl is not None else stop_loss,
   322	            tp=tp if tp is not None else take_profit,

    71	    def process_tick(
    72	        self,
    73	        symbol: str,
    74	        price: float,
    75	        quantity: float,
    76	        strategy_signal: Optional[str] = None,
    77	        extra_market_data: Optional[Dict[str, Any]] = None,
    78	    ) -> Dict[str, Any]:
    79	        """
    80	        Process incoming tick data safely through the entire simulation stack.
    81	        """
    82	        sym = str(symbol).strip().upper()
    83	        px = float(price)
    84	        qty = float(quantity)
    85	
    86	        # 1. Update mark-to-market equity and calculate drawdown
    87	        current_equity = self.simulator.mark_to_market({sym: px})
    88	        current_dd = self.portfolio_manager.calculate_drawdown(current_equity)
    89	        rec_status = self.recovery_manager.evaluate(current_dd)
    90	
    91	        # 2. Check if recovery manager or emergency breaker blocks trades
    92	        recovery_locked = self.recovery_manager.get_status().get("is_active", False)
    93	
    94	        # 3. Prepare pipeline input
    95	        payload: Dict[str, Any] = {
    96	            "symbol": sym,
    97	            "price": px,
    98	            "quantity": qty,
    99	            "action": strategy_signal or "HOLD",
   100	        }
   101	        if extra_market_data:
   102	            payload.update(extra_market_data)
   103	
   104	        # 4. Process cycle in TradingPipeline
   105	        pipe_res = self.pipeline.process_cycle(payload)
   106	
   107	        # Resolve pipeline decision
   108	        pipe_decision = str(pipe_res.get("decision", "HOLD")).upper()
   109	        pipe_signal = str(pipe_res.get("signal") or pipe_res.get("action") or "HOLD").upper()
   110	
   111	        is_blocked = (
   112	            recovery_locked
   113	            or "SAFE_HOLD" in pipe_decision
   114	            or "RECOVERY" in pipe_decision
   115	            or "EMERGENCY" in pipe_decision
   116	            or pipe_signal == "HOLD"
   117	        )
   118	
   119	        if is_blocked:
   120	            record = {
   121	                "timestamp": self._timestamp(),
   122	                "symbol": sym,
   123	                "action": "HOLD",
   124	                "pipeline_decision": pipe_res,
   125	                "recovery_status": rec_status,
   126	                "order_submitted": False,
   127	                "real_trading_enabled": False,
   128	                "simulated": True,
   129	            }
   130	            self.execution_log.append(record)
   131	            return record
   132	
   133	        action = pipe_signal if pipe_signal in {"BUY", "SELL"} else "HOLD"
   134	        if action == "HOLD":
   135	            record = {
   136	                "timestamp": self._timestamp(),
   137	                "symbol": sym,
   138	                "action": "HOLD",
   139	                "pipeline_decision": pipe_res,
   140	                "recovery_status": rec_status,
   141	                "order_submitted": False,
   142	                "real_trading_enabled": False,
   143	                "simulated": True,
   144	            }
   145	            self.execution_log.append(record)
   146	            return record
   147	
   148	        # 5. Virtual order execution via simulator
   149	        sim_res = self.simulator.execute(
   150	            symbol=sym,
   151	            side=action,
   152	            quantity=qty,
   153	            price=px,
   154	        )
   155	
   156	        # 6. Mirror executed trade in PortfolioManager
   157	        if sim_res.get("status") == "filled":
   158	            fee = sim_res.get("fee", 0.0)
   159	            if action == "BUY":
   160	                self.portfolio_manager.record_buy(sym, qty, px, fee)
   161	            elif action == "SELL":
   162	                self.portfolio_manager.record_sell(sym, qty, px, fee)
   163	
   164	        updated_equity = self.simulator.mark_to_market({sym: px})
   165	
   166	        result = {
   167	            "timestamp": self._timestamp(),
   168	            "symbol": sym,
   169	            "action": action,
   170	            "pipeline_decision": pipe_res,
   171	            "execution": sim_res,
   172	            "equity": updated_equity,
   173	            "order_submitted": False,
   174	            "real_trading_enabled": False,
   175	            "simulated": True,
   176	        }
   177	        self.execution_log.append(result)
   178	        return result
   179

   181	        """Aggregate complete system status."""

   201	    breaker = getattr(self.risk_manager, "circuit_breaker_tripped", False)
   202	    if callable(breaker):
   203	        breaker = breaker()
   204	
   205	    if breaker:
   206	        return {"action": "HOLD", "blocked": True}
   207	
   208	    if not callable(getattr(self, "strategy", None)):
   209	        return {"action": "HOLD", "status": "NO_STRATEGY"}
   210	
   211	    decision = self.strategy(market_data) or {}
   212	    action = str(decision.get("action", "HOLD")).upper()
   213	
   214	    if action not in ("BUY", "SELL"):
   215	        return {"action": "HOLD"}
   216	
   217	    symbol = str(decision.get("symbol", "BTCUSDT")).upper()
   218	    quantity = float(decision.get("quantity", decision.get("size", 0.0)))
   219	
   220	    price = decision.get("price", decision.get("current_price"))
   221	    if price is None:
   222	        tick = market_data.get(symbol, {})
   223	        if isinstance(tick, dict):
   224	            price = tick.get("price", tick.get("close"))
   225	
   226	    if price is None or float(price) <= 0 or quantity <= 0:
   227	        return {"action": "HOLD", "blocked": True, "reason": "INVALID_SIGNAL"}
   228	
   229	    result = self.order_executor.place_and_execute_market_order(
   230	        symbol=symbol,
   231	        side=action,

   233	        current_price=float(price),

   235	        tp=decision.get("tp", decision.get("take_profit")),
   236	    )
   237	
   238	    return {
   239	        "action": action,
   240	        "order": result["order"],
   241	        "position": result.get("position"),
   242	        "status": result.get("status", "FILLED"),
   243	    }
