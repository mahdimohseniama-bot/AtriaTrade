     2	"""
     3	AtriaTrade Web App
     4	- Serves the dashboard UI
     5	- Keeps current backend APIs
     6	- Paper-trading safe only
     7	"""
     8	
     9	from __future__ import annotations
    10	
    11	import os
    12	import time
    13	from typing import Any, Optional
    14	
    15	import uvicorn
    16	from fastapi import Depends, FastAPI, HTTPException, Request
    17	from fastapi.responses import HTMLResponse
    18	from fastapi.security import APIKeyHeader
    19	from fastapi.staticfiles import StaticFiles
    20	from fastapi.templating import Jinja2Templates
    21	from pydantic import BaseModel, Field
    22	
    23	# ---------------------------------------------------------------------
    24	# Project imports
    25	# ---------------------------------------------------------------------
    26	try:
    27	    from src.dashboard.server import DashboardServer
    28	    from src.dashboard.auth import SecureAuthManager
    29	except ImportError as e:
    30	    print(f"❌ ImportError: Could not import dashboard components. {e}")
    31	    print("   Please ensure:")
    32	    print("   1. You are running this script from the root directory of AtriaTrade (~/AtriaTrade).")
    33	    print("   2. The directory structure includes src/dashboard/server.py and src/dashboard/auth.py.")
    34	    raise
    35	
    36	# ---------------------------------------------------------------------
    37	# Configuration
    38	# ---------------------------------------------------------------------
    39	AUTH_TOKEN_HEADER_NAME = "X-Auth-Token"
    40	ADMIN_TOKEN_FOR_TEST = "SET_YOUR_ADMIN_TOKEN_HERE"  # replace with a real token if needed
    41	
    42	WEB_DIR = os.path.dirname(os.path.abspath(__file__))
    43	TEMPLATES_DIR = os.path.join(WEB_DIR, "templates")
    44	STATIC_DIR = os.path.join(WEB_DIR, "static")
    45	
    46	# ---------------------------------------------------------------------
    47	# Auth / dashboard backend
    48	# ---------------------------------------------------------------------
    49	auth_manager = SecureAuthManager()
    50	dashboard_server = DashboardServer(auth_manager=auth_manager)
    51	
    52	# ---------------------------------------------------------------------
    53	# FastAPI app
    54	# ---------------------------------------------------------------------
    55	app = FastAPI(
    56	    title="AtriaTrade Dashboard API",
    57	    description="API for managing and monitoring the AtriaTrade bot. Paper-trading safe.",
    58	    version="1.2.0",
    59	)
    60	
    61	api_key_header = APIKeyHeader(name=AUTH_TOKEN_HEADER_NAME, auto_error=True)
    62	
    63	templates = Jinja2Templates(directory=TEMPLATES_DIR)
    64	
    65	if os.path.isdir(STATIC_DIR):
    66	    app.mount("/static", StaticFiles(directory=STATIC_DIR), name="static")
    67	
    68	
    69	# ---------------------------------------------------------------------
    70	# Helpers
    71	# ---------------------------------------------------------------------
    72	def get_client_ip(request: Request) -> str:
    73	    if request.client and request.client.host:
    74	        return request.client.host
    75	
    76	    forwarded_for = request.headers.get("X-Forwarded-For")
    77	    if forwarded_for:
    78	        return forwarded_for.split(",")[0].strip()
    79	
    80	    return "127.0.0.1"
    81	
    82	
    83	def require_auth_token(token: str) -> None:
    84	    if token == ADMIN_TOKEN_FOR_TEST:
    85	        return
    86	    # If you later generate dynamic tokens with SecureAuthManager,
    87	    # the dashboard_server layer can validate them.
    88	    # This fallback keeps local development simple.
    89	    raise HTTPException(
    90	        status_code=401,
    91	        detail={"message": "Unauthorized. Invalid or expired token."},
    92	    )
    93

    95	# ---------------------------------------------------------------------
    96	# Models
    97	# ---------------------------------------------------------------------
    98	class PublicStatusResponse(BaseModel):
    99	    system: str = Field(..., description="System name")
   100	    online: bool = Field(..., description="System online state")
   101	    timestamp: float = Field(..., description="Unix time")
   102	
   103	
   104	class BotStateData(BaseModel):
   105	    status: str
   106	    mode: str
   107	    active_positions: int
   108	    total_balance_usdt: float
   109	    profit_reserved: float
   110	
   111	
   112	class DashboardSummaryResponse(BaseModel):
   113	    success: bool
   114	    user: Optional[str] = None
   115	    role: Optional[str] = None
   116	    data: Optional[BotStateData] = None
   117	    error: Optional[str] = None
   118	
   119	
   120	class UpdateBotStateResponse(BaseModel):
   121	    success: bool
   122	    updated_key: Optional[str] = None
   123	    new_value: Optional[Any] = None
   124	    error: Optional[str] = None
   125	
   126	
   127	# ---------------------------------------------------------------------
   128	# UI routes
   129	# ---------------------------------------------------------------------
   130	@app.get("/", response_class=HTMLResponse)
   131	async def root(request: Request):
   132	    return templates.TemplateResponse(
   133	        "dashboard.html",
   134	        {"request": request, "page_title": "AtriaTrade Dashboard"},
   135	    )
   136	
   137	
   138	@app.get("/dashboard", response_class=HTMLResponse)
   139	async def dashboard_page(request: Request):
   140	    return templates.TemplateResponse(
   141	        "dashboard.html",
   142	        {"request": request, "page_title": "AtriaTrade Dashboard"},
   143	    )
   144	
   145	
   146	# ---------------------------------------------------------------------
   147	# API routes
   148	# ---------------------------------------------------------------------
   149	@app.get(
   150	    "/api/public/status",
   151	    response_model=PublicStatusResponse,
   152	    summary="Get Public System Status",
   153	    tags=["Public Endpoints"],
   154	)
   155	async def get_public_status_endpoint(request: Request):
   156	    return PublicStatusResponse(
   157	        system="AtriaTrade API",
   158	        online=True,
   159	        timestamp=time.time(),
   160	    )
   161	
   162	
   163	@app.get(
   164	    "/api/dashboard/summary",
   165	    response_model=DashboardSummaryResponse,
   166	    summary="Get Authenticated Dashboard Summary",
   167	    tags=["Dashboard API"],
   168	    dependencies=[Depends(api_key_header)],
   169	)
   170	async def get_dashboard_summary_endpoint(
   171	    request: Request,
   172	    token: str = Depends(api_key_header),
   173	):
   174	    client_ip = get_client_ip(request)
   175	
   176	    if token != ADMIN_TOKEN_FOR_TEST:
   177	        # Try backend validation if a real token system is available
   178	        result = await dashboard_server.get_dashboard_summary(
   179	            token=token,
   180	            client_ip=client_ip,
   181	        )
   182	        if not result.get("success", False):
   183	            raise HTTPException(
   184	                status_code=401,
   185	                detail={"message": result.get("error", "Authentication failed.")},
   186	            )
   187	
   188	        bot_data = None
   189	        if result.get("data"):
   190	            try:
   191	                bot_data = BotStateData(**result["data"])
   192	            except Exception as e:
   193	                raise HTTPException(
   194	                    status_code=500,
   195	                    detail={"message": f"Failed to parse dashboard data: {e}"},
   196	                )
   197	
   198	        return DashboardSummaryResponse(
   199	            success=True,
   200	            user=result.get("user"),
   201	            role=result.get("role"),
   202	            data=bot_data,
   203	        )
   204	
   205	    # Local dev fallback
   206	    return DashboardSummaryResponse(
   207	        success=True,
   208	        user="admin",
   209	        role="admin",
   210	        data=BotStateData(
   211	            status="RUNNING",
   212	            mode="PAPER_TRADING",
   213	            active_positions=0,
   214	            total_balance_usdt=0.0,
   215	            profit_reserved=0.0,
   216	        ),
   217	    )
   218	
   219	
   220	@app.put(
   221	    "/api/bot/state",
   222	    response_model=UpdateBotStateResponse,
   223	    summary="Update Bot State Parameter",
   224	    tags=["Admin Control"],
   225	    dependencies=[Depends(api_key_header)],
   226	)
   227	async def update_bot_state_endpoint(
   228	    request: Request,
   229	    key: str,
   230	    value: Any,
   231	    token: str = Depends(api_key_header),
   232	):
   233	    if token != ADMIN_TOKEN_FOR_TEST:
   234	        client_ip = get_client_ip(request)
   235	        result = await dashboard_server.update_bot_state(
   236	            key=key,
   237	            value=value,
   238	            token=token,
   239	            client_ip=client_ip,
   240	        )
   241	        if not result.get("success", False):
   242	            raise HTTPException(
   243	                status_code=403,
   244	                detail={"message": result.get("error", "Unknown error during state update.")},
   245	            )
   246	        return UpdateBotStateResponse(
   247	            success=True,
   248	            updated_key=result.get("updated_key"),
   249	            new_value=result.get("new_value"),
   250	        )
   251	
   252	    # Safe local fallback: do not change live trading state
   253	    return UpdateBotStateResponse(
   254	        success=True,
   255	        updated_key=key,
   256	        new_value=value,
   257	    )
