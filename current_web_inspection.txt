    77	# --- Utility Functions ---
    78	def get_client_ip(request: Request) -> str:
    79	    """
    80	    Retrieves the client's IP address from the request.
    81	    Handles common proxy headers like X-Forwarded-For for accurate IP logging.
    82	    """
    83	    if request.client and request.client.host:
    84	        return request.client.host
    85	    
    86	    forwarded_for = request.headers.get("X-Forwarded-For")
    87	    if forwarded_for:
    88	        return forwarded_for.split(",")[0].strip()
    89	        
    90	    return "127.0.0.1"

    95	class PublicStatusResponse(BaseModel):
    96	    """Response model for the public status endpoint."""
    97	    system: str = Field(..., description="Name of the system component (e.g., 'AtriaTrade API').")

   101	class BotStateData(BaseModel):
   102	    """Detailed data model for the bot's current operational state."""
   103	    status: str = Field(..., description="Current status of the bot (e.g., RUNNING, STOPPED, PAUSED, ERROR).")
   104	    mode: str = Field(..., description="Trading mode (e.g., PAPER_TRADING, TESTNET, LIVE_TRADING).")

   109	class DashboardSummaryResponse(BaseModel):
   110	    """Comprehensive response model for the authenticated dashboard summary."""
   111	    success: bool = Field(..., description="Indicates if the request was processed successfully.")
   112	    user: Optional[str] = Field(None, description="Username of the authenticated user.")
   113	    role: Optional[str] = Field(None, description="Role assigned to the user (e.g., 'admin', 'trader', 'viewer').")
   114	    data: Optional[BotStateData] = Field(None, description="Detailed bot state information, included only on success.")
   115	    error: Optional[str] = Field(None, description="Error message if the operation failed (e.g., authentication, authorization).")
   116	
   117	class UpdateBotStateResponse(BaseModel):

   143	@app.get(
   144	    "/api/dashboard/summary",
   145	    response_model=DashboardSummaryResponse,
   146	    summary="Get Authenticated Dashboard Summary",
   147	    tags=["Dashboard API"],
   148	    dependencies=[Depends(api_key_header)],
   149	    description="Retrieves detailed bot status, user information, and trading data. Requires a valid authentication token."
   150	)
   151	async def get_dashboard_summary_endpoint(
   152	    request: Request,
   153	    token: str = Depends(api_key_header),
   154	):
   155	    """
   156	    Fetches the main dashboard summary data. Requires authentication via `X-Auth-Token` header.
   157	    Returns user details and bot operational state if the token is valid.
   158	    """
   159	    client_ip = get_client_ip(request)
   160	    result = await dashboard_server.get_dashboard_summary(token=token, client_ip=client_ip)
   161	
   162	    if not result.get("success", False):
   163	        raise HTTPException(
   164	            status_code=401,
   165	            detail={"message": result.get("error", "Authentication failed. Invalid or expired token.")}
   166	        )
   167	
   168	    bot_data = None
   169	    if "data" in result and result["data"]:
   170	        try:
   171	            bot_data = BotStateData(**result["data"])
   172	        except Exception as e:
   173	            print(f"Error parsing BotStateData: {e}")
   174	            raise HTTPException(
   175	                status_code=500,
   176	                detail={"message": "Internal server error processing dashboard data."}
   177	            )
   178	
   179	    return DashboardSummaryResponse(
   180	        success=True,
   181	        user=result.get("user"),
   182	        role=result.get("role"),
   183	        data=bot_data
   184	    )
   185	
   186	@app.put(
   187	    "/api/bot/state",
   188	    response_model=UpdateBotStateResponse,
   189	    summary="Update Bot State Parameter (Admin Only)",
   190	    tags=["Admin Control"],
   191	    dependencies=[Depends(api_key_header)],
   192	    description="Allows authenticated administrators to modify specific bot state parameters dynamically. Requires 'admin' role."
   193	)
   194	async def update_bot_state_endpoint(
   195	    request: Request,
   196	    key: str,
   197	    value: Any,
   198	    token: str = Depends(api_key_header),
   199	):
   200	    """
   201	    Endpoint to update a specific parameter in the bot's state.
   202	    Requires authentication and the user must have admin privileges.
   203	    Example Usage: PUT /api/bot/state?key=mode&value="LIVE_TRADING"
   204	    """
   205	    client_ip = get_client_ip(request)
   206	    result = await dashboard_server.update_bot_state(key=key, value=value, token=token, client_ip=client_ip)
   207	
   208	    if not result.get("success", False):
   209	        error_message = result.get("error", "Unknown error during state update.")
   210	        status_code = 403
   211	
   212	        if "Unauthorized" in error_message or "Invalid token" in error_message:
   213	            status_code = 401
   214	        elif "Not an administrator" in error_message or "permission denied" in error_message:
   215	            status_code = 403
   216	        else:
   217	            status_code = 500
   218	
   219	        raise HTTPException(status_code=status_code, detail={"message": error_message})
   220	
   221	    return UpdateBotStateResponse(
   222	        success=True,
   223	        updated_key=result.get("updated_key"),
   224	        new_value=result.get("new_value")
   225	    )
