     1	"""Trading Engine module for orchestrating signals, execution, and portfolio management."""
     2	from typing import Dict, Any, Optional
     3	import time
     4	
     5	
     6	class TradingEngine:
     7	    def __init__(
     8	        self,
     9	        portfolio_manager: Optional[Any] = None,
    10	        risk_manager: Optional[Any] = None,
    11	        order_executor: Optional[Any] = None,
    12	        profit_reserve_manager: Optional[Any] = None,
    13	        **kwargs
    14	    ):
    15	        # نگاشت سازگار برای پارامترهای تستی و ماژولار
    16	        self.portfolio_manager = portfolio_manager or kwargs.get("portfolio")
    17	        self.risk_manager = risk_manager or kwargs.get("risk")
    18	        self.order_executor = order_executor or kwargs.get("executor")
    19	        self.profit_reserve_manager = profit_reserve_manager or kwargs.get("reserve")
    20	
    21	        self.is_running = False
    22	        self.last_tick_time = 0.0
    23	        self._entry_prices: Dict[str, float] = {}
    24	
    25	    def start(self) -> None:
    26	        self.is_running = True
    27	
    28	    def stop(self) -> None:
    29	        self.is_running = False
    30	
    31	    def process_tick(self, tick: Dict[str, Any]) -> Optional[Dict[str, Any]]:
    32	        """Processes an incoming market tick and executes simulated paper trades."""
    33	        if not self.is_running:
    34	            return None
    35	
    36	        if not isinstance(tick, dict):
    37	            return None
    38	
    39	        symbol = tick.get("symbol")
    40	        price = tick.get("price")
    41	        signal = tick.get("signal")
    42	        confidence = tick.get("confidence", 1.0)
    43	
    44	        if not symbol or price is None or price <= 0:
    45	            return None
    46	
    47	        if signal not in ["BUY", "SELL", "HOLD"]:
    48	            return None
    49	
    50	        if signal == "HOLD":
    51	            return None
    52	
    53	        self.last_tick_time = time.time()
    54	
    55	        if signal == "BUY":
    56	            qty = tick.get("quantity") or tick.get("qty")
    57	            if not qty:
    58	                # محاسبه حجم پیش‌فرض بر اساس پورتفولیو
    59	                if self.portfolio_manager and hasattr(self.portfolio_manager, "cash"):
    60	                    alloc_cash = self.portfolio_manager.cash * 0.5
    61	                    qty = alloc_cash / price
    62	                else:
    63	                    qty = 0.1
    64	
    65	            # بررسی و ثبت خرید
    66	            if self.portfolio_manager:
    67	                if hasattr(self.portfolio_manager, "can_allocate"):
    68	                    if not self.portfolio_manager.can_allocate(symbol, qty * price):
    69	                        return None
    70	
    71	                buy_res = {}
    72	                if hasattr(self.portfolio_manager, "record_buy"):
    73	                    buy_res = self.portfolio_manager.record_buy(symbol=symbol, quantity=qty, price=price) or {}
    74	                if isinstance(buy_res, dict) and buy_res.get("status") == "REJECTED":
    75	                    return None
    76	
    77	            self._entry_prices[symbol] = float(price)
    78	
    79	            return {
    80	                "symbol": symbol,
    81	                "side": "BUY",
    82	                "price": price,
    83	                "quantity": qty,
    84	                "status": "FILLED",
    85	                "timestamp": self.last_tick_time
    86	            }
    87	
    88	        elif signal == "SELL":
    89	            current_qty = 0.0
    90	            if self.portfolio_manager:
    91	                if hasattr(self.portfolio_manager, "get_position"):
    92	                    current_qty = self.portfolio_manager.get_position(symbol)
    93	                elif hasattr(self.portfolio_manager, "positions"):
    94	                    pos = self.portfolio_manager.positions.get(symbol, {})
    95	                    if isinstance(pos, dict):
    96	                        current_qty = pos.get("quantity", 0.0)
    97	                    else:
    98	                        current_qty = getattr(pos, "quantity", 0.0)
    99	
   100	            if current_qty <= 0.0:
   101	                return None
   102	
   103	            qty_to_sell = tick.get("quantity") or tick.get("qty") or current_qty
   104	            qty_to_sell = min(qty_to_sell, current_qty)
   105	
   106	            sell_res = {}
   107	            if self.portfolio_manager and hasattr(self.portfolio_manager, "record_sell"):
   108	                sell_res = self.portfolio_manager.record_sell(symbol=symbol, quantity=qty_to_sell, price=price) or {}
   109	
   110	            # محاسبه سود محقق‌شده
   111	            realized_pnl = 0.0
   112	            if isinstance(sell_res, dict) and "realized_pnl" in sell_res:
   113	                realized_pnl = float(sell_res["realized_pnl"])
   114	            else:
   115	                entry_p = self._entry_prices.get(symbol, price)
   116	                realized_pnl = (float(price) - float(entry_p)) * float(qty_to_sell)
   117	
   118	            # اگر از طریق محاسبات پورتفولیو هم سود مثبت بود ولی صفر گزارش شده بود:
   119	            if realized_pnl <= 0.0 and symbol in self._entry_prices:
   120	                entry_p = self._entry_prices[symbol]
   121	                if price > entry_p:
   122	                    realized_pnl = (float(price) - float(entry_p)) * float(qty_to_sell)
   123	
   124	            # انتقال سود به ProfitReserveManager
   125	            if realized_pnl > 0.0 and self.profit_reserve_manager:
   126	                reserve = self.profit_reserve_manager
   127	                if hasattr(reserve, "add_to_vault") and callable(reserve.add_to_vault):
   128	                    reserve.add_to_vault(realized_pnl)
   129	                elif hasattr(reserve, "process_trade_profit") and callable(reserve.process_trade_profit):
   130	                    reserve.process_trade_profit(realized_pnl)
   131	                elif hasattr(reserve, "process_trade_pnl") and callable(reserve.process_trade_pnl):
   132	                    reserve.process_trade_pnl(realized_pnl)
   133	                elif hasattr(reserve, "deposit") and callable(reserve.deposit):
   134	                    reserve.deposit(realized_pnl)
   135	                elif hasattr(reserve, "record_profit") and callable(reserve.record_profit):
   136	                    reserve.record_profit(realized_pnl)
   137	                else:
   138	                    cur_v = getattr(reserve, "vault_balance", 0.0) or 0.0
   139	                    setattr(reserve, "vault_balance", float(cur_v) + realized_pnl)
   140	
   141	            return {
   142	                "symbol": symbol,
   143	                "side": "SELL",
   144	                "price": price,
   145	                "quantity": qty_to_sell,
   146	                "realized_pnl": realized_pnl,
   147	                "status": "FILLED",
   148	                "timestamp": self.last_tick_time
   149	            }
   150	
   151	        return None

     1	import logging
     2	from typing import Dict, Any, Optional
     3	
     4	logger = logging.getLogger(__name__)
     5	
     6	class PortfolioManager:
     7	    def __init__(self, initial_cash: float = 10000.0, max_asset_weight: float = 0.5, **kwargs):
     8	        self.cash: float = float(kwargs.get("initial_balance", initial_cash))
     9	        self.max_asset_weight: float = float(max_asset_weight)
    10	        self.positions: Dict[str, Dict[str, float]] = {}
    11	        self.peak_equity: float = self.cash
    12	        self.equity: float = self.cash
    13	        self.drawdown: float = 0.0
    14	        self.realized_pnl: float = 0.0
    15	        self.trade_count: int = 0
    16	
    17	    @property
    18	    def holdings(self) -> Dict[str, Dict[str, float]]:
    19	        return self.positions
    20	
    21	    def get_balance(self) -> float:
    22	        return self.cash
    23	
    24	    def get_position(self, symbol: str) -> float:
    25	        pos = self.positions.get(symbol, 0.0)
    26	        if isinstance(pos, dict):
    27	            return float(pos.get("quantity", 0.0))
    28	        return float(pos)
    29	
    30	    def get_holding_quantity(self, symbol: str) -> float:
    31	        return self.get_position(symbol)
    32	
    33	    def get_equity(self, current_prices: Optional[Dict[str, float]] = None) -> float:
    34	        if current_prices is None:
    35	            current_prices = {}
    36	        total = self.cash
    37	        for sym, pos in self.positions.items():
    38	            qty = pos.get("quantity", 0.0) if isinstance(pos, dict) else float(pos)
    39	            price = current_prices.get(sym, pos.get("avg_price", 0.0) if isinstance(pos, dict) else 0.0)
    40	            total += qty * price
    41	        self.equity = total
    42	        if total > self.peak_equity:
    43	            self.peak_equity = total
    44	        self.calculate_drawdown(self.equity)
    45	        return self.equity
    46	
    47	    def calculate_total_equity(self, current_prices: Optional[Dict[str, float]] = None) -> float:
    48	        return self.get_equity(current_prices)
    49	
    50	    def calculate_drawdown(self, equity_val: Optional[Any] = None) -> float:
    51	        if isinstance(equity_val, (int, float)):
    52	            current_eq = float(equity_val)
    53	        elif isinstance(equity_val, dict):
    54	            current_eq = self.get_equity(equity_val)
    55	        else:
    56	            current_eq = self.equity
    57	
    58	        if self.peak_equity <= 0:
    59	            self.drawdown = 0.0
    60	            return 0.0
    61	
    62	        dd = ((self.peak_equity - current_eq) / self.peak_equity) * 100.0
    63	        self.drawdown = max(0.0, dd)
    64	        return self.drawdown
    65	
    66	    def get_summary(self, current_prices: Optional[Dict[str, float]] = None) -> Dict[str, Any]:
    67	        equity = self.get_equity(current_prices)
    68	        dd_pct = self.calculate_drawdown(equity)
    69	        return {
    70	            "cash": self.cash,
    71	            "total_equity": equity,
    72	            "equity": equity,
    73	            "drawdown": dd_pct / 100.0,
    74	            "drawdown_pct": dd_pct,
    75	            "peak_equity": self.peak_equity,
    76	            "realized_pnl": self.realized_pnl,
    77	            "trade_count": self.trade_count,
    78	            "positions": self.positions
    79	        }
    80	
    81	    def can_allocate(self, symbol: str, amount: float, current_prices: Optional[Dict[str, float]] = None) -> Dict[str, Any]:
    82	        equity = self.get_equity(current_prices)
    83	        max_allowed = equity * self.max_asset_weight
    84	        current_holding_val = 0.0
    85	        if symbol in self.positions:
    86	            pos = self.positions[symbol]
    87	            qty = pos.get("quantity", 0.0) if isinstance(pos, dict) else float(pos)
    88	            px = (current_prices or {}).get(symbol, pos.get("avg_price", 0.0) if isinstance(pos, dict) else 0.0)
    89	            current_holding_val = qty * px
    90	
    91	        if (current_holding_val + amount) > max_allowed:
    92	            return {
    93	                "allowed": False,
    94	                "reason": f"Exceeds max asset weight limit ({self.max_asset_weight * 100}%)"
    95	            }
    96	        if amount > self.cash:
    97	            return {
    98	                "allowed": False,
    99	                "reason": "Insufficient cash balance"
   100	            }
   101	        return {"allowed": True, "reason": "OK"}

   103	    def record_buy(self, symbol: str, quantity: float, price: float, fee: float = 0.0) -> Dict[str, Any]:
   104	        total_cost = (quantity * price) + fee
   105	        if self.cash < total_cost:
   106	            return {
   107	                "status": "REJECTED",
   108	                "reason": "Insufficient cash to cover cost and fees"
   109	            }
   110	
   111	        self.cash -= total_cost
   112	        if symbol not in self.positions:
   113	            self.positions[symbol] = {"quantity": quantity, "avg_price": price}
   114	        else:
   115	            current_pos = self.positions[symbol]
   116	            old_qty = current_pos["quantity"]
   117	            old_avg = current_pos["avg_price"]
   118	            new_qty = old_qty + quantity
   119	            new_avg = ((old_qty * old_avg) + (quantity * price)) / new_qty if new_qty > 0 else price
   120	            self.positions[symbol] = {"quantity": new_qty, "avg_price": new_avg}
   121	
   122	        self.trade_count += 1
   123	        return {
   124	            "status": "FILLED",
   125	            "action": "BUY",
   126	            "symbol": symbol,
   127	            "quantity": quantity,
   128	            "price": price,
   129	            "fee": fee,
   130	            "total_cost": total_cost,
   131	            "cost": total_cost,
   132	            "cash": self.cash
   133	        }

   135	    def record_sell(self, symbol: str, quantity: float, price: float, fee: float = 0.0) -> Dict[str, Any]:
   136	        current_qty = self.get_position(symbol)
   137	        if current_qty < quantity or quantity <= 0:
   138	            return {
   139	                "status": "REJECTED",
   140	                "reason": "Insufficient position quantity to sell"
   141	            }
   142	
   143	        pos = self.positions[symbol]
   144	        avg_price = pos["avg_price"]
   145	        pnl = (price - avg_price) * quantity - fee
   146	
   147	        proceeds = (quantity * price) - fee
   148	        self.cash += proceeds
   149	        self.realized_pnl += pnl
   150	
   151	        remaining_qty = current_qty - quantity
   152	        if remaining_qty <= 1e-8:
   153	            self.positions.pop(symbol, None)
   154	        else:
   155	            self.positions[symbol]["quantity"] = remaining_qty
   156	
   157	        self.trade_count += 1
   158	        return {
   159	            "status": "FILLED",
   160	            "action": "SELL",
   161	            "symbol": symbol,
   162	            "quantity": quantity,
   163	            "price": price,
   164	            "fee": fee,
   165	            "realized_pnl": pnl,
   166	            "cash": self.cash
